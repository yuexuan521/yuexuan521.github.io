---

**格式** 

```
输入格式: 
输入正整数N

输出格式: 
输出YES或者NO
```

```
输入格式: 
1234006

输出格式: 
YES
```

***（2）参考代码*** 

```c
#include<stdio.h>
int main() 
{ 
    int N;
    scanf("%d", &N);
    int temp = N, count = 0, sum1 = 0, sum2 = 0;
    while (temp)
    {
        temp /= 10;
        count++;
    }
    if (count == 1)
    {
        printf("YES");
    }
    else
    {
        for (int i=1;i<=count;i++)
        {
            if (i < (count - 1) / 2 + 1)
            {
                sum1 += N % 10;
            }
            else if (i > (count - 1) / 2 + 1)
            {
                sum2 += N % 10;
            }
            N /= 10;
        }
    }

    {
```

```c
#include<stdio.h>

{
```

```c
    scanf("%d %d %d %d %d %d", &x1, &y1, &x2, &y2, &x3, &y3);
    scanf("%d %d", &xN, &yN);
    double S, S1, S2, S3;
    S = (double)(x1*y2+x2*y3+x3*y1-x1*y3-x2*y1-x3*y2) * 0.5;
```
