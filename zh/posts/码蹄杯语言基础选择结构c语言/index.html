#include<stdio.h>
int main()
{
    int a, b, c;
    scanf("%d %d %d", &a, &b, &c);
    if ((a > b && b > c) || (a < b && b < c))
    {
        printf("%d", b);
    }
    else if ((c > a && a > b) || (c < a && a < b))
    {
        printf("%d", a);
    }
    else
    {
        printf("%d", c);
    }
    return 0;
}

#include<stdio.h>
#include<ctype.h>
int main()
{
    char str;
    scanf("%c", &str);
    if (isalpha(str))
    {
        if (str >= 65 && str <= 90)
        {
            str += 32;
            printf("%c", str);
        }
        else
        {
            str -= 32;
            printf("%c", str);
        }
    }
    else
    {
        printf("%c", str);
    }
    return 0;
}

⭐MT1121小码哥考完咯#

#include<stdio.h>
int main()
{
    int grade;
    scanf("%d", &grade);
    grade = grade / 10;
    switch (grade)
    {
    case 10:
        printf("A");
        break;
    case 9:
        printf("A");
        break;
    case 8:
        printf("B");
        break;
    case 7:
        printf("C");
        break;
    case 6:
        printf("D");
        break;
    default:
        printf("F");
    }
    return 0;
}
