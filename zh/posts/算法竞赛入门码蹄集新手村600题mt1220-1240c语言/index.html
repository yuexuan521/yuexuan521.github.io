#include<stdio.h>
int main() 
{ 
    int a, b, n1, n2;
    scanf("%d %d", &a, &b);
    double sum = 0, num = 0;
    for (int i=1;;i++)
    {
        num += (1.0 / (double)i);
        sum += 1.0 / num;
        if (a < sum)
        {
            n1 = i;
            break;
        }
    }
    for (int i=n1+1;;i++)
    {
        num += (1.0 / (double)i);
        sum += 1.0 / num;
        if (b <= sum)
        {
            n2 = i - 1;
            break;
        }
    }
    printf("%d %d", n1, n2);
    return 0; 
}

#include<stdio.h>
int main()
{
    double x = 0, PI = 0;
    for (int i = 1;;i += 2)
    {
        x = 1 / (double)i;
        if (x < 1e-6)
        {
            break;
        }
        if (i % 4 == 3)
        {
            x = -x;
        }
        PI += x;
    }
    printf("%.2lf\n", 4 * PI);
    return 0;
}
