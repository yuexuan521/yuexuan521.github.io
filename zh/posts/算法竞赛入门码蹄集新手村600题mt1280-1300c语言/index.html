        }

            printf("%d ", Disarium);
        }
    }
    return 0; 
}

#include<stdio.h>
int main() 
{ 
    int A[100] = {1, 1}, S[100] = {1, 2};
    for (int i=2;;i++)
    {
        A[i] = A[i-2] + 2 * A[i - 1];
        S[i] = S[i - 1] + A[i];
        if (S[i-1] < 100 && S[i] >= 100)
        {

            printf("%d ", i);
        }

            printf("%d", i);

        }
    }

}
