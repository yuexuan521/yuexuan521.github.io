#include<stdio.h>
//正、负实数强制转换成整数后，被截断。例如，（2.3）--》（2），（-1.2）--》（-1）
int getceil(double x)
{
    if (x > 0)
    {
        if (x - (int)x > 0)
        {
            return (int)x + 1;
        }
        else
        {
            return (int)x;
        }
    }
    else
    {
        return (int)x;
    }
}
int main()
{
    double x;
    scanf("%lf", &x);
    printf("%d\n", getceil(x));
    return 0;
}

#include<stdio.h>
int getfloor(double x)
{
    if (x > 0)
    {
        return (int)x;
    }
    else
    {
        if (x - (int)x < 0)
        {
            return (int)x - 1;
        }
        else
        {
            return (int)x;
        }
    }
}
int main()
{
    double x;
    scanf("%lf", &x);
    printf("%d\n", getfloor(x));
    return 0;
}

#include<stdio.h>
int fact(int x)
{
    int i, sum = 1;
    for (i = 1; i <= x; i++)
    {
        sum *= i;
    }
    return sum;

    return 0;

#include<stdio.h>
int fun(int m, int n)
{
    int i, result = 1;
    for (i = 0; i < n; i++)
    {
        result *= m;
    }
    return result;
}
int main()
{
    int m, n;

    return 0;
}
