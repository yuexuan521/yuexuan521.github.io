    return 0; 
}

（1）题目 输入一个数字N，把偶数位置的所有数更改为0。比如30，二进制形式是11110,替换后是01010，则输出对应的十进制整数10。最右边是第0位，从右向左依次为0位，1位等等。不考虑非法输入等特殊情况。

格式

输入为整型

输出格式: 
输出为整型

样例1

#include<stdio.h>
int main() 
{ 
    int N;
    scanf("%d", &N);
    int ID = 0, ch = 1, temp = N;
    while (temp)
    {
        if (ID % 2 == 0 && temp % 2 == 1)
        {
            N = N - ch;
        }
        temp = temp / 2;
        ch = 2 * ch;
        ID++;
    }
    printf("%d\n", N);
    return 0; 
}
