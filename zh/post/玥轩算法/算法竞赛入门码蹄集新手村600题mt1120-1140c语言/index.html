int main() 
{
    int grade;
    scanf("%d", &grade);
    grade = grade / 10;
    switch(grade)
    {
    case 10:
        printf("A");
        break;
    case 9:
        printf("A");
        break;
    case 8:
        printf("B");
        break;
    case 7:
        printf("C");

        printf("D");
        break;
    default :
        printf("F");
    }
    return 0; 
}

        printf("Error");
    }
    return 0; 
}

    {
        printf("Number");
    }
    else if ((int)str >= 65 && (int)str <= 90)

    {
        printf("Logical operators");
    }
    else
    {
        printf("Other character");
    }
    return 0; 
}
