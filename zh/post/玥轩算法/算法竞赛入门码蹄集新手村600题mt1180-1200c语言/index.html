            printf("NO");

        {

        }
        else
        {
            printf("NO");
        }
    }
    return 0; 
}

{
    int n, i;
    double temp, p_sum = 0, n_sum = 0;
    scanf("%d", &n);
    for (i=0;i<n;i++)
    {
