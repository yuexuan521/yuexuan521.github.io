#include<stdio.h>
int main() 
{ 
    int N, factorial, count = 0;
    scanf("%d", &N);
    for (int i=1;i<=N;i++)
    {
        factorial = 1;
        for (int j=1;j<=i;j++)
        {
            factorial *= j;
        }
        if (factorial % 10 == 0)
        {
            count++;
        }
    }
    printf("%d", count);
    return 0; 
}
